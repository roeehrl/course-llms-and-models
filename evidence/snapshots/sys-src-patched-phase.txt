"""Short-lived, generation-fenced workers for non-accelerator epoch work.

The registered miner remains the sole coordinator/P2P owner.  This module
isolates deterministic CPU blob construction in a spawned process so the
controller can park its MPS model before network upload without moving wallet
or attestation authority across the process boundary.

Only immutable file inputs and an epoch binding cross the boundary.  A child
may produce bytes, but the parent accepts them only after validating a
0600 receipt, exact binding, byte count, and SHA-256 digest.
"""

from __future__ import annotations

import asyncio
import gc
import hashlib
import json
import math
import mmap
import multiprocessing
import os
import re
import stat
import tempfile
import time
import uuid
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any

import torch

from subnet.utils.blob_format import build_weights_blob
from subnet.utils.vector_utils import all_finite
from common.utils.s3_utils import upload_parts


_SCHEMA_VERSION = 1
_COPY_CHUNK_BYTES = 8 * 1024 * 1024
_IDENTIFIER_RE = re.compile(r"^[A-Za-z0-9._:-]{1,160}$")
_UPLOAD_JOB_NAME = "upload-job.json"
_UPLOAD_RECEIPT_NAME = "upload-receipt.json"
_JOB_WAIT_POLL_SECONDS = 0.05


class PhaseWorkerError(RuntimeError):
    """The isolated phase worker failed or returned untrusted output."""


@dataclass(frozen=True, slots=True)
class PhaseBinding:
    run_id: str
    layer: int
    epoch: int
    generation: int
    task_id: str

    @classmethod
    def create(cls, *, run_id: str, layer: int, epoch: int, generation: int) -> "PhaseBinding":
        return cls(
            run_id=run_id,
            layer=layer,
            epoch=epoch,
            generation=generation,
            task_id=uuid.uuid4().hex,
        )

    @classmethod
    def from_dict(cls, value: object) -> "PhaseBinding":
        if not isinstance(value, dict):
            raise PhaseWorkerError("phase binding must be an object")
        try:
            binding = cls(
                run_id=value["run_id"],
                layer=value["layer"],
                epoch=value["epoch"],
                generation=value["generation"],
                task_id=value["task_id"],
            )
        except (KeyError, TypeError) as exc:
            raise PhaseWorkerError("phase binding is incomplete") from exc
        binding.validate()
        return binding

    def validate(self) -> None:
        if not isinstance(self.run_id, str) or not _IDENTIFIER_RE.fullmatch(self.run_id):
            raise PhaseWorkerError("phase binding run_id is invalid")
        if not isinstance(self.task_id, str) or not re.fullmatch(r"[0-9a-f]{32}", self.task_id):
            raise PhaseWorkerError("phase binding task_id is invalid")
        for name, value in (
            ("layer", self.layer),
            ("epoch", self.epoch),
            ("generation", self.generation),
        ):
            if isinstance(value, bool) or not isinstance(value, int) or value < 0:
                raise PhaseWorkerError(f"phase binding {name} is invalid")


@dataclass(frozen=True, slots=True)
class PhaseArtifact:
    binding: PhaseBinding
    path: Path
    sha256: str
    byte_count: int
    pseudo_mean: float
    pseudo_std: float
    previous_mean: float
    previous_std: float
    current_mean: float
    current_std: float
    clipped: bool
    worker_pid: int
    elapsed_seconds: float


class MMapSliceBuffer:
    """Read-only, sliceable mmap accepted by the existing multipart uploader."""

    def __init__(self, path: Path):
        flags = os.O_RDONLY
        if hasattr(os, "O_NOFOLLOW"):
            flags |= os.O_NOFOLLOW
        self._fd = os.open(path, flags)
        try:
            file_stat = os.fstat(self._fd)
            if not stat.S_ISREG(file_stat.st_mode) or file_stat.st_size <= 0:
                raise PhaseWorkerError("phase artifact is not a non-empty regular file")
            self._length = file_stat.st_size
            self._mapping = mmap.mmap(self._fd, self._length, access=mmap.ACCESS_READ)
        except BaseException:
            os.close(self._fd)
            raise

    def __len__(self) -> int:
        return self._length

    def __getitem__(self, item: slice) -> memoryview:
        if not isinstance(item, slice):
            raise TypeError("MMapSliceBuffer only supports slice indexing")
        start, stop, step = item.indices(self._length)
        if step != 1:
            raise ValueError("MMapSliceBuffer does not support stepped slices")
        return memoryview(self._mapping)[start:stop]

    def close(self) -> None:
        try:
            try:
                self._mapping.close()
            except BufferError:
                # The upload has already finished; a slice of a part whose
                # request failed (broken pipe) or was cancelled (hedged
                # duplicate) can still be referenced by aiohttp internals or a
                # traceback. Collect cycles and retry once, then leave the
                # read-only mapping for process exit instead of turning a
                # completed upload into a failed one (patch 27).
                gc.collect()
                try:
                    self._mapping.close()
                except BufferError:
                    pass
        finally:
            os.close(self._fd)

    def __enter__(self) -> "MMapSliceBuffer":
        return self

    def __exit__(self, _exc_type, _exc, _tb) -> None:
        self.close()


class PhaseWorkspace:
    """A private spool directory whose complete lifetime is parent-owned."""

    def __init__(self, root: Path, binding: PhaseBinding):
        binding.validate()
        root.mkdir(parents=True, exist_ok=True, mode=0o700)
        os.chmod(root, 0o700)
        self.path = Path(tempfile.mkdtemp(prefix=f"epoch-{binding.epoch}-", dir=root))
        os.chmod(self.path, 0o700)
        self.binding = binding

    def write_tensor(self, name: str, tensor: torch.Tensor) -> Path:
        if not _IDENTIFIER_RE.fullmatch(name):
            raise PhaseWorkerError("phase tensor name is invalid")
        target = self.path / f"{name}.pt"
        temporary = self.path / f".{name}.{uuid.uuid4().hex}.tmp"
        with temporary.open("xb") as handle:
            os.chmod(temporary, 0o600)
            torch.save(tensor.detach().to("cpu").contiguous(), handle)
            handle.flush()
            os.fsync(handle.fileno())
        os.replace(temporary, target)
        return target

    def cleanup(self) -> None:
        if not self.path.exists():
            return
        for child in self.path.iterdir():
            if child.is_file() and not child.is_symlink():
                child.unlink()
            else:
                raise PhaseWorkerError(f"refusing unsafe phase-spool cleanup entry: {child.name}")
        self.path.rmdir()

    def __enter__(self) -> "PhaseWorkspace":
        return self

    def __exit__(self, _exc_type, _exc, _tb) -> None:
        self.cleanup()


def _read_json(path: Path) -> dict[str, Any]:
    with path.open("r", encoding="utf-8") as handle:
        value = json.load(handle)
    if not isinstance(value, dict):
        raise PhaseWorkerError(f"{path.name} must contain a JSON object")
    return value


def _write_json_atomic(path: Path, value: dict[str, Any]) -> None:
    temporary = path.with_name(f".{path.name}.{uuid.uuid4().hex}.tmp")
    with temporary.open("x", encoding="utf-8") as handle:
        os.chmod(temporary, 0o600)
        json.dump(value, handle, sort_keys=True, separators=(",", ":"))
        handle.write("\n")
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)


def _sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb", buffering=0) as handle:
        while chunk := handle.read(_COPY_CHUNK_BYTES):
            digest.update(chunk)
    return digest.hexdigest()


def _write_buffer_atomic(path: Path, buffer: object) -> tuple[int, str]:
    temporary = path.with_name(f".{path.name}.{uuid.uuid4().hex}.tmp")
    digest = hashlib.sha256()
    byte_count = len(buffer)  # type: ignore[arg-type]
    with temporary.open("xb", buffering=0) as handle:
        os.chmod(temporary, 0o600)
        for start in range(0, byte_count, _COPY_CHUNK_BYTES):
            chunk = buffer[start : min(start + _COPY_CHUNK_BYTES, byte_count)]  # type: ignore[index]
            handle.write(chunk)
            digest.update(chunk)
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)
    return byte_count, digest.hexdigest()


def _require_finite(tensor: torch.Tensor, label: str) -> None:
    if tensor.numel() == 0:
        raise PhaseWorkerError(f"{label} is empty")
    # Chunked (patch 37): a whole-tensor isfinite on the 1 GB weights needed
    # about 2.5 GB of masks and an abs copy in this child.
    if not all_finite(tensor):
        raise PhaseWorkerError(f"{label} contains NaN or Inf")


def _load_tensor(path: Path, label: str) -> torch.Tensor:
    flags = os.O_RDONLY
    if hasattr(os, "O_NOFOLLOW"):
        flags |= os.O_NOFOLLOW
    fd = os.open(path, flags)
    try:
        file_stat = os.fstat(fd)
        if not stat.S_ISREG(file_stat.st_mode) or file_stat.st_size <= 0:
            raise PhaseWorkerError(f"{label} input is not a non-empty regular file")
    finally:
        os.close(fd)
    try:
        tensor = torch.load(path, map_location="cpu", mmap=True, weights_only=True)
    except TypeError:  # PyTorch before weights_only support.
        tensor = torch.load(path, map_location="cpu", mmap=True)
    if not isinstance(tensor, torch.Tensor):
        raise PhaseWorkerError(f"{label} input is not a tensor")
    tensor = tensor.detach().to("cpu").contiguous()
    _require_finite(tensor, label)
    return tensor


def _build_pseudogradient(job: dict[str, Any]) -> tuple[torch.Tensor, dict[str, float | bool]]:
    previous = _load_tensor(Path(job["previous_weights_path"]), "previous weights")
    current = _load_tensor(Path(job["current_weights_path"]), "current weights")
    if previous.shape != current.shape:
        raise PhaseWorkerError(
            f"weight shape mismatch: previous={tuple(previous.shape)} current={tuple(current.shape)}"
        )

    batch_count = job["pseudo_gradient_batch_count"]
    if isinstance(batch_count, bool) or not isinstance(batch_count, int) or batch_count <= 0:
        raise PhaseWorkerError("pseudo_gradient_batch_count must be positive")
    pseudo = torch.empty_like(previous, dtype=torch.bfloat16, device="cpu")
    chunk_size = -(-previous.numel() // batch_count)
    for start in range(0, previous.numel(), chunk_size):
        end = min(start + chunk_size, previous.numel())
        difference = previous[start:end].to(torch.float32) - current[start:end].to(torch.float32)
        pseudo[start:end] = difference.to(torch.bfloat16)

    clipped = False
    clip_max_norm = job.get("clip_max_norm")
    if clip_max_norm is not None:
        if isinstance(clip_max_norm, bool) or not isinstance(clip_max_norm, (int, float)):
            raise PhaseWorkerError("clip_max_norm must be a finite positive number")
        clip_max_norm = float(clip_max_norm)
        if not math.isfinite(clip_max_norm) or clip_max_norm <= 0:
            raise PhaseWorkerError("clip_max_norm must be a finite positive number")
        current_norm = pseudo.norm(2).item()
        if current_norm > clip_max_norm:
            pseudo = pseudo * (clip_max_norm / (current_norm + 1e-6))
            clipped = True

    _require_finite(pseudo, "pseudo gradients")
    statistics: dict[str, float | bool] = {
        "pseudo_mean": float(pseudo.mean().item()),
        "pseudo_std": float(pseudo.std().item()),
        "previous_mean": float(previous.mean().item()),
        "previous_std": float(previous.std().item()),
        "current_mean": float(current.mean().item()),
        "current_std": float(current.std().item()),
        "clipped": clipped,
    }
    return pseudo, statistics


def _wait_for_job(job_path: Path, timeout_seconds: float) -> None:
    """Block a pre-spawned child until the parent publishes its job file.

    The parent writes jobs atomically (temporary file + ``os.replace``), so a
    visible job file is always complete. A child that never receives a job
    exits non-zero without a receipt; the parent treats that like any other
    worker failure and falls back to a fresh spawn.
    """
    deadline = time.monotonic() + max(0.0, float(timeout_seconds))
    while not job_path.exists():
        if time.monotonic() >= deadline:
            raise PhaseWorkerError(
                f"pre-spawned phase worker received no job within {float(timeout_seconds):.0f}s"
            )
        time.sleep(_JOB_WAIT_POLL_SECONDS)


def _phase_worker_main(job_path_string: str, wait_for_job_seconds: float = 0.0) -> None:
    job_path = Path(job_path_string)
    receipt_path: Path | None = None
    binding: PhaseBinding | None = None
    if wait_for_job_seconds > 0:
        # STRATA critical path: the interpreter, torch and the S3 client were
        # imported while the parent was still building the blob. Elapsed time
        # in the receipt must describe the upload, not the wait.
        _wait_for_job(job_path, wait_for_job_seconds)
    started_at = time.monotonic()
    try:
        job = _read_json(job_path)
        if job.get("schema_version") != _SCHEMA_VERSION:
            raise PhaseWorkerError("unsupported phase-worker job")
        binding = PhaseBinding.from_dict(job.get("binding"))
        receipt_path = Path(job["receipt_path"])
        if receipt_path.exists():
            raise PhaseWorkerError("phase-worker output already exists")
        operation = job.get("operation")
        if operation == "weights_blob":
            output_path = Path(job["output_path"])
            if output_path.exists():
                raise PhaseWorkerError("phase-worker output already exists")
            pseudo, statistics = _build_pseudogradient(job)
            num_sections = job["num_sections"]
            local_steps = job["local_optimization_steps"]
            if isinstance(num_sections, bool) or not isinstance(num_sections, int) or num_sections <= 0:
                raise PhaseWorkerError("num_sections must be positive")
            if isinstance(local_steps, bool) or not isinstance(local_steps, int) or local_steps <= 0:
                raise PhaseWorkerError("local_optimization_steps must be positive")
            buffer, trailer = build_weights_blob(
                pseudo,
                num_sections=num_sections,
                local_optimization_steps=local_steps,
            )
            byte_count, digest = _write_buffer_atomic(output_path, buffer)
            receipt = {
                "schema_version": _SCHEMA_VERSION,
                "status": "complete",
                "operation": operation,
                "binding": asdict(binding),
                "artifact": {
                    "path": str(output_path),
                    "sha256": digest,
                    "byte_count": byte_count,
                },
                "statistics": statistics,
                "trailer": trailer,
                "worker_pid": os.getpid(),
                "elapsed_seconds": time.monotonic() - started_at,
            }
        elif operation == "s3_upload":
            artifact_path = Path(job["artifact_path"])
            expected_digest = job["artifact_sha256"]
            if _sha256_file(artifact_path) != expected_digest:
                raise PhaseWorkerError("phase artifact changed before isolated upload")
            dns_cache_path = job.get("dns_cache_path")
            if dns_cache_path:
                # Patch 43: storage hosts resolved by an earlier upload back
                # this one if the resolver fails right now.
                try:
                    from miner import settings as miner_settings
                    from miner.utils import dns_fallback

                    if dns_fallback.install(miner_settings.DNS_SERVE_STALE_MAX_SECONDS):
                        dns_fallback.load_file(Path(dns_cache_path))
                    else:
                        dns_cache_path = None
                except Exception:
                    dns_cache_path = None
            with MMapSliceBuffer(artifact_path) as payload:
                # Multipart paths remain mmap-backed. A single-part upload is
                # necessarily below the multipart threshold, so materialize
                # that small payload to avoid retaining an exported memoryview
                # when the mapping closes after aiohttp releases its request.
                upload_payload = payload if len(job["urls"]) > 1 else bytes(payload[0 : len(payload)])
                try:
                    parts = asyncio.run(
                        upload_parts(
                            urls=job["urls"],
                            data=upload_payload,
                            upload_id=job.get("upload_id"),
                        )
                    )
                finally:
                    if dns_cache_path:
                        from miner.utils import dns_fallback

                        dns_fallback.save_file(Path(dns_cache_path))
            if parts is None:
                parts = []
            # The pre-upload digest above already bound the mmap contents to
            # the receipt digest; the workspace is parent-owned and 0700, so a
            # second full read here only delays the coordinator call that
            # closes the upload window. Keep the cheap size check.
            expected_bytes = job.get("artifact_bytes")
            if isinstance(expected_bytes, int) and artifact_path.stat().st_size != expected_bytes:
                raise PhaseWorkerError("phase artifact changed during isolated upload")
            receipt = {
                "schema_version": _SCHEMA_VERSION,
                "status": "complete",
                "operation": operation,
                "binding": asdict(binding),
                "artifact_sha256": expected_digest,
                "parts": parts,
                "worker_pid": os.getpid(),
                "elapsed_seconds": time.monotonic() - started_at,
            }
        else:
            raise PhaseWorkerError("unsupported phase-worker operation")
        _write_json_atomic(receipt_path, receipt)
    except BaseException as exc:
        if receipt_path is not None and binding is not None and not receipt_path.exists():
            try:
                _write_json_atomic(
                    receipt_path,
                    {
                        "schema_version": _SCHEMA_VERSION,
                        "status": "error",
                        "binding": asdict(binding),
                        "error_type": type(exc).__name__,
                        "error": str(exc)[:500],
                        "worker_pid": os.getpid(),
                        "elapsed_seconds": time.monotonic() - started_at,
                    },
                )
            except BaseException:
                pass
        raise


def _validate_artifact(
    *,
    expected_binding: PhaseBinding,
    expected_path: Path,
    receipt_path: Path,
) -> PhaseArtifact:
    receipt = _read_json(receipt_path)
    if receipt.get("schema_version") != _SCHEMA_VERSION or receipt.get("status") != "complete":
        error_type = receipt.get("error_type", "PhaseWorkerError")
        error = receipt.get("error", "worker did not complete")
        raise PhaseWorkerError(f"isolated phase worker failed: {error_type}: {error}")
    actual_binding = PhaseBinding.from_dict(receipt.get("binding"))
    if actual_binding != expected_binding:
        raise PhaseWorkerError("isolated phase worker returned a stale or mismatched epoch binding")

    artifact = receipt.get("artifact")
    statistics = receipt.get("statistics")
    if not isinstance(artifact, dict) or not isinstance(statistics, dict):
        raise PhaseWorkerError("isolated phase worker receipt is incomplete")
    if Path(artifact.get("path", "")) != expected_path:
        raise PhaseWorkerError("isolated phase worker returned an unexpected artifact path")
    expected_digest = artifact.get("sha256")
    expected_bytes = artifact.get("byte_count")
    if not isinstance(expected_digest, str) or not re.fullmatch(r"[0-9a-f]{64}", expected_digest):
        raise PhaseWorkerError("isolated phase worker returned an invalid digest")
    if isinstance(expected_bytes, bool) or not isinstance(expected_bytes, int) or expected_bytes <= 0:
        raise PhaseWorkerError("isolated phase worker returned an invalid byte count")
    file_stat = expected_path.lstat()
    if not stat.S_ISREG(file_stat.st_mode) or file_stat.st_size != expected_bytes:
        raise PhaseWorkerError("isolated phase artifact type or byte count is invalid")
    if _sha256_file(expected_path) != expected_digest:
        raise PhaseWorkerError("isolated phase artifact digest mismatch")

    return PhaseArtifact(
        binding=actual_binding,
        path=expected_path,
        sha256=expected_digest,
        byte_count=expected_bytes,
        pseudo_mean=float(statistics["pseudo_mean"]),
        pseudo_std=float(statistics["pseudo_std"]),
        previous_mean=float(statistics["previous_mean"]),
        previous_std=float(statistics["previous_std"]),
        current_mean=float(statistics["current_mean"]),
        current_std=float(statistics["current_std"]),
        clipped=bool(statistics["clipped"]),
        worker_pid=int(receipt["worker_pid"]),
        elapsed_seconds=float(receipt["elapsed_seconds"]),
    )


async def prepare_weights_blob_isolated(
    *,
    workspace: PhaseWorkspace,
    current_weights: torch.Tensor,
    previous_weights_path: Path,
    num_sections: int,
    local_optimization_steps: int,
    pseudo_gradient_batch_count: int,
    clip_max_norm: float | None,
    timeout_seconds: float,
) -> PhaseArtifact:
    """Build a wire-identical weights blob in a spawn-isolated CPU worker."""

    binding = workspace.binding
    binding.validate()
    current_path = workspace.write_tensor("current-weights", current_weights)
    output_path = workspace.path / "weights.blob"
    receipt_path = workspace.path / "receipt.json"
    job_path = workspace.path / "job.json"
    _write_json_atomic(
        job_path,
        {
            "schema_version": _SCHEMA_VERSION,
            "operation": "weights_blob",
            "binding": asdict(binding),
            "current_weights_path": str(current_path),
            "previous_weights_path": str(previous_weights_path),
            "output_path": str(output_path),
            "receipt_path": str(receipt_path),
            "num_sections": num_sections,
            "local_optimization_steps": local_optimization_steps,
            "pseudo_gradient_batch_count": pseudo_gradient_batch_count,
            "clip_max_norm": clip_max_norm,
        },
    )

    context = multiprocessing.get_context("spawn")
    process = context.Process(
        target=_phase_worker_main,
        args=(str(job_path),),
        name=f"IOTAPhaseWorker-e{binding.epoch}-g{binding.generation}",
        daemon=False,
    )
    process.start()
    try:
        await asyncio.to_thread(process.join, timeout_seconds)
        if process.is_alive():
            raise PhaseWorkerError(
                f"isolated phase worker timed out after {timeout_seconds:.1f}s "
                f"for epoch {binding.epoch} generation {binding.generation}"
            )
    finally:
        if process.is_alive():
            process.terminate()
            await asyncio.to_thread(process.join, 5.0)
            if process.is_alive():
                process.kill()
                await asyncio.to_thread(process.join, 5.0)
    if process.exitcode != 0:
        if receipt_path.exists():
            receipt = _read_json(receipt_path)
            raise PhaseWorkerError(
                f"isolated phase worker exited {process.exitcode}: "
                f"{receipt.get('error_type', 'error')}: {receipt.get('error', 'unknown error')}"
            )
        raise PhaseWorkerError(f"isolated phase worker exited {process.exitcode} without a receipt")
    return _validate_artifact(
        expected_binding=binding,
        expected_path=output_path,
        receipt_path=receipt_path,
    )


def start_upload_worker_early(
    *,
    workspace: PhaseWorkspace,
    timeout_seconds: float,
) -> multiprocessing.Process:
    """Spawn the upload child before its job exists so its startup overlaps the build.

    A spawned child pays for interpreter start plus torch and aiohttp imports
    (about two seconds on the Studio). Starting it while the CPU worker is
    still building the blob takes that cost off the upload window's critical
    path. The child blocks in ``_wait_for_job`` until ``upload_artifact_isolated``
    publishes the job atomically; if the parent never does (build failure), it
    terminates the child, and a child that outlives ``timeout_seconds`` exits
    on its own.
    """
    job_path = workspace.path / _UPLOAD_JOB_NAME
    if job_path.exists():
        raise PhaseWorkerError("upload job was published before the pre-spawned worker started")
    context = multiprocessing.get_context("spawn")
    process = context.Process(
        target=_phase_worker_main,
        args=(str(job_path), float(timeout_seconds)),
        name=f"IOTAUploadWorker-e{workspace.binding.epoch}-g{workspace.binding.generation}-prespawn",
        daemon=False,
    )
    process.start()
    return process


def terminate_worker(process: multiprocessing.Process | None) -> None:
    """Stop a pre-spawned worker that will never receive a job."""
    if process is None or not process.is_alive():
        return
    process.terminate()
    process.join(5.0)
    if process.is_alive():
        process.kill()
        process.join(5.0)


async def upload_artifact_isolated(
    *,
    workspace: PhaseWorkspace,
    artifact: PhaseArtifact,
    urls: list[str],
    upload_id: str | None,
    timeout_seconds: float,
    process: multiprocessing.Process | None = None,
) -> tuple[list[dict[str, Any]], int, float, str]:
    """Upload an immutable artifact in a separate, credential-free process.

    ``process`` may be a worker from ``start_upload_worker_early``; it is used
    when still alive, otherwise a fresh worker is spawned. The returned
    ``dispatch`` names which path ran.
    """

    if artifact.binding != workspace.binding:
        raise PhaseWorkerError("upload artifact binding does not match workspace")
    # ``_validate_artifact`` hashed this file moments ago and the child hashes
    # it again before the first PUT; a third full read here only delays the
    # upload. Size is verified cheaply and the digest is bound in the job.
    if artifact.path.stat().st_size != artifact.byte_count:
        raise PhaseWorkerError("phase artifact changed before upload dispatch")
    if not urls:
        raise PhaseWorkerError("isolated upload requires at least one URL")

    receipt_path = workspace.path / _UPLOAD_RECEIPT_NAME
    job_path = workspace.path / _UPLOAD_JOB_NAME
    _write_json_atomic(
        job_path,
        {
            "schema_version": _SCHEMA_VERSION,
            "operation": "s3_upload",
            "binding": asdict(artifact.binding),
            "receipt_path": str(receipt_path),
            "artifact_path": str(artifact.path),
            "artifact_sha256": artifact.sha256,
            "artifact_bytes": artifact.byte_count,
            "urls": urls,
            "upload_id": upload_id,
            # Patch 43: answers shared by upload processes across epochs.
            "dns_cache_path": str(workspace.path.parent / "dns-cache.json"),
        },
    )

    dispatch = "spawned"
    if process is not None:
        if process.is_alive():
            dispatch = "pre-spawned"
        else:
            dispatch = "respawned-after-prespawn-exit"
            process = None
    if process is None:
        context = multiprocessing.get_context("spawn")
        process = context.Process(
            target=_phase_worker_main,
            args=(str(job_path),),
            name=f"IOTAUploadWorker-e{artifact.binding.epoch}-g{artifact.binding.generation}",
            daemon=False,
        )
        process.start()
    try:
        await asyncio.to_thread(process.join, timeout_seconds)
        if process.is_alive():
            raise PhaseWorkerError(
                f"isolated upload timed out after {timeout_seconds:.1f}s "
                f"for epoch {artifact.binding.epoch} generation {artifact.binding.generation}"
            )
    finally:
        if process.is_alive():
            process.terminate()
            await asyncio.to_thread(process.join, 5.0)
            if process.is_alive():
                process.kill()
                await asyncio.to_thread(process.join, 5.0)

    if process.exitcode != 0:
        if receipt_path.exists():
            receipt = _read_json(receipt_path)
            raise PhaseWorkerError(
                f"isolated upload exited {process.exitcode}: "
                f"{receipt.get('error_type', 'error')}: {receipt.get('error', 'unknown error')}"
            )
        raise PhaseWorkerError(f"isolated upload exited {process.exitcode} without a receipt")

    receipt = _read_json(receipt_path)
    if (
        receipt.get("schema_version") != _SCHEMA_VERSION
        or receipt.get("status") != "complete"
        or receipt.get("operation") != "s3_upload"
        or PhaseBinding.from_dict(receipt.get("binding")) != artifact.binding
        or receipt.get("artifact_sha256") != artifact.sha256
    ):
        raise PhaseWorkerError("isolated upload returned a stale or invalid receipt")
    parts = receipt.get("parts")
    if not isinstance(parts, list) or any(not isinstance(part, dict) for part in parts):
        raise PhaseWorkerError("isolated upload returned invalid multipart metadata")
    return parts, int(receipt["worker_pid"]), float(receipt["elapsed_seconds"]), dispatch


def move_optimizer_state_(optimizer: torch.optim.Optimizer, device: str | torch.device) -> int:
    """Move every tensor owned by an optimizer state dictionary in place."""

    target = torch.device(device)
    moved = 0
    for state_values in optimizer.state.values():
        for key, value in tuple(state_values.items()):
            if isinstance(value, torch.Tensor) and value.device != target:
                state_values[key] = value.to(target)
                moved += 1
    return moved


def park_compute_plane(model_manager: object) -> dict[str, int | str | bool]:
    """Move model/optimizer tensors to CPU and release accelerator caches."""

    model = getattr(model_manager, "model", None)
    optimizer = getattr(model_manager, "optimizer", None)
    configured_device = str(getattr(model_manager, "device", "cpu"))
    if model is None or optimizer is None:
        raise PhaseWorkerError("cannot park an uninitialized compute plane")
    if getattr(model_manager, "_phase_compute_parked", False):
        return {"parked": True, "configured_device": configured_device, "optimizer_tensors_moved": 0}

    mps_backend = getattr(getattr(torch, "backends", None), "mps", None)
    mps_available = bool(mps_backend is not None and mps_backend.is_available())
    if configured_device.startswith("mps") and mps_available:
        torch.mps.synchronize()
    model.to("cpu")
    moved = move_optimizer_state_(optimizer, "cpu")
    for parameter in model.parameters():
        if parameter.device.type != "cpu":
            raise PhaseWorkerError("model parameter remained on the accelerator while parking")
    setattr(model_manager, "_phase_compute_parked", True)
    if configured_device.startswith("mps") and mps_available:
        torch.mps.empty_cache()
    elif configured_device.startswith("cuda") and torch.cuda.is_available():
        torch.cuda.empty_cache()
    return {"parked": True, "configured_device": configured_device, "optimizer_tensors_moved": moved}


def unpark_compute_plane(model_manager: object) -> dict[str, int | str | bool]:
    """Restore a parked model/optimizer to its configured accelerator."""

    model = getattr(model_manager, "model", None)
    optimizer = getattr(model_manager, "optimizer", None)
    configured_device = str(getattr(model_manager, "device", "cpu"))
    if model is None or optimizer is None:
        raise PhaseWorkerError("cannot unpark an uninitialized compute plane")
    if not getattr(model_manager, "_phase_compute_parked", False):
        return {"parked": False, "configured_device": configured_device, "optimizer_tensors_moved": 0}

    model.to(configured_device)
    moved = move_optimizer_state_(optimizer, configured_device)
    target_type = torch.device(configured_device).type
    for parameter in model.parameters():
        if parameter.device.type != target_type:
            raise PhaseWorkerError("model parameter remained off the configured accelerator while unparking")
    setattr(model_manager, "_phase_compute_parked", False)
    return {"parked": False, "configured_device": configured_device, "optimizer_tensors_moved": moved}
