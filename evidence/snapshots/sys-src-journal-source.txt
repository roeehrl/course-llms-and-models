"""Owner-only structured event journal for the local STRATA app.

The mounted proxy creates ``events-<session>.jsonl`` (0600) and passes its
path in ``STRATA_IOTA_EVENT_JOURNAL``. The worker appends one JSON object per
domain event at the exact code sites that also log the event, so the app can
read typed fields instead of scraping free-text log lines whose wording and
truncation can change. When the variable is unset (upstream behaviour,
standalone runs) every call is a no-op.

Records: ``schema_version`` (1), ``wall_time`` (UTC ISO 8601),
``monotonic_seconds``, ``event`` and the event's fields. Values are JSON
primitives; strings are bounded. Appends use one ``os.write`` on an
``O_APPEND`` descriptor, like the registration journal. A journal failure is
logged at most once a minute and never changes worker behaviour.
"""

from __future__ import annotations

import json
import os
import time
from datetime import datetime, timezone

from loguru import logger

SCHEMA_VERSION = 1
ENVIRONMENT_VARIABLE = "STRATA_IOTA_EVENT_JOURNAL"
MAX_STRING_LENGTH = 2048
MAX_RECORD_BYTES = 64 * 1024

_last_failure_log = 0.0


def _bounded(value, depth: int = 0):
    if value is None or isinstance(value, (bool, int)):
        return value
    if isinstance(value, float):
        return value if value == value and value not in (float("inf"), float("-inf")) else None
    if isinstance(value, str):
        return value[:MAX_STRING_LENGTH]
    if depth < 3 and isinstance(value, dict):
        return {str(key)[:128]: _bounded(item, depth + 1) for key, item in list(value.items())[:64]}
    if depth < 3 and isinstance(value, (list, tuple)):
        return [_bounded(item, depth + 1) for item in list(value)[:64]]
    return str(value)[:MAX_STRING_LENGTH]


def _emit_unchecked(event: str, fields: dict) -> None:
    path = os.environ.get(ENVIRONMENT_VARIABLE)
    if not path:
        return
    record = {
        "schema_version": SCHEMA_VERSION,
        "wall_time": datetime.now(timezone.utc).isoformat(),
        "monotonic_seconds": time.monotonic(),
        "event": event,
    }
    for key, value in fields.items():
        if key not in record:
            record[key] = _bounded(value)
    payload = (json.dumps(record, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode("utf-8")
    if len(payload) > MAX_RECORD_BYTES:
        return
    flags = os.O_WRONLY | os.O_APPEND | os.O_CREAT | getattr(os, "O_NOFOLLOW", 0) | getattr(os, "O_CLOEXEC", 0)
    descriptor = os.open(path, flags, 0o600)
    try:
        os.fchmod(descriptor, 0o600)
        os.write(descriptor, payload)
    finally:
        os.close(descriptor)


def emit(event: str, **fields) -> None:
    """Append one event; best effort, never raises."""
    global _last_failure_log
    try:
        _emit_unchecked(event, fields)
    except Exception as exc:  # noqa: BLE001 - diagnostics must not affect training
        now = time.monotonic()
        if now - _last_failure_log >= 60:
            _last_failure_log = now
            logger.warning(
                "Local event journal append failed (event={}, error_type={})",
                event,
                type(exc).__name__,
            )
